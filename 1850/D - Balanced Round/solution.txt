t = int(input())
for _ in range(t):
    n, k = map(int,input().split())
    q = list(map(int,input().split()))
    q.sort()
    i = 0
    maxx = 0
    while i < n:
        count = 1
        while i < n - 1 and abs(q[i] - q[i+1]) <= k:
            i+=1
            count += 1
        i+=1
        maxx = max(maxx,count)
    print(n - maxx)
